        // ============================================
        // SKYLINE: First-Person Flight Game
        // Enhanced with dramatic outdoor visuals

        const CONFIG = {
            graphics: 'medium',
            cloudQuality: 1.0,
            shadowQuality: 1.0,
            mouseSensitivity: 1.0,
            masterVolume: 0.5,
            engineVolume: 0.5
        };

        // THREE.JS SCENE SETUP
        const scene = new THREE.Scene();
        const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 500000);
        const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: false });

        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.setPixelRatio(window.devicePixelRatio);
        renderer.setClearColor(0x87ceeb);
        document.body.appendChild(renderer.domElement);

        // FLIGHT STATE
        const flightState = {
            position: new THREE.Vector3(0, 5000, 0),
            velocity: new THREE.Vector3(0, 0, 0),
            
            pitch: 0,
            roll: 0,
            yaw: 0,
            
            pitchRate: 0,
            rollRate: 0,
            yawRate: 0,
            
            airspeed: 0,
            maxAirspeed: 500,
            
            throttle: 0,
            maxThrottle: 100,
            
            altitude: 5000,
            verticalSpeed: 0,
            heading: 0,
            
            engineRunning: true,
            vibrationIntensity: 0,
            turbulenceIntensity: 0
        };

        // INPUT HANDLING
        const keys = {};
        let mouseX = 0.5, mouseY = 0.5;
        let mouseInWindow = true;

        window.addEventListener('keydown', (e) => {
            keys[e.key.toLowerCase()] = true;
            if (e.key === 'Escape') {
                e.preventDefault();
                togglePause();
            }
        });

        window.addEventListener('keyup', (e) => {
            keys[e.key.toLowerCase()] = false;
        });

        window.addEventListener('mousemove', (e) => {
            mouseX = e.clientX / window.innerWidth;
            mouseY = e.clientY / window.innerHeight;
        });

        window.addEventListener('mouseenter', () => { mouseInWindow = true; });
        window.addEventListener('mouseleave', () => { mouseInWindow = false; });

        // CAMERA LOOK CONTROL
        let camPitch = 0, camYaw = 0;

        function updateCameraLook() {
            if (!mouseInWindow) return;

            const sensitivity = CONFIG.mouseSensitivity * 0.003;
            camYaw -= (mouseX - 0.5) * 2 * sensitivity * 100;
            camPitch += (mouseY - 0.5) * 2 * sensitivity * 100;

            camPitch = Math.max(-Math.PI / 2, Math.min(Math.PI / 2, camPitch));
        }

        // FLIGHT PHYSICS ENGINE
        function updateFlightPhysics(deltaTime) {
            if (flightState.engineRunning) {
                if (keys['shift']) {
                    flightState.throttle = Math.min(flightState.throttle + 0.5 * deltaTime, 100);
                }
                if (keys['control'] || keys['ctrl']) {
                    flightState.throttle = Math.max(flightState.throttle - 0.5 * deltaTime, 0);
                }
            }

            // Control inputs with smooth damping
            const controlResponse = 0.05;
            if (keys['w']) flightState.pitchRate = Math.min(flightState.pitchRate + controlResponse, 1);
            if (keys['s']) flightState.pitchRate = Math.max(flightState.pitchRate - controlResponse, -1);
            if (keys['a']) flightState.rollRate = Math.min(flightState.rollRate + controlResponse, 1);
            if (keys['d']) flightState.rollRate = Math.max(flightState.rollRate - controlResponse, -1);
            if (keys['q']) flightState.yawRate = Math.min(flightState.yawRate + controlResponse * 0.5, 0.5);
            if (keys['e']) flightState.yawRate = Math.max(flightState.yawRate - controlResponse * 0.5, -0.5);

            flightState.pitchRate *= 0.92;
            flightState.rollRate *= 0.92;
            flightState.yawRate *= 0.92;

            flightState.pitch += flightState.pitchRate * 0.02;
            flightState.roll += flightState.rollRate * 0.02;
            flightState.yaw += flightState.yawRate * 0.01;

            flightState.pitch = Math.max(-Math.PI * 0.5, Math.min(Math.PI * 0.5, flightState.pitch));

            let dragMultiplier = 1;
            if (keys[' ']) {
                dragMultiplier = 0.6;
                flightState.airspeed *= 0.97;
            }

            const targetAirspeed = (flightState.throttle / 100) * flightState.maxAirspeed;
            flightState.airspeed += (targetAirspeed - flightState.airspeed) * 0.02 * dragMultiplier;

            // Build flight vectors
            const forward = new THREE.Vector3(
                Math.sin(flightState.yaw) * Math.cos(flightState.pitch),
                Math.sin(flightState.pitch),
                Math.cos(flightState.yaw) * Math.cos(flightState.pitch)
            ).normalize();

            const right = new THREE.Vector3(
                Math.cos(flightState.yaw + Math.PI / 2),
                0,
                Math.sin(flightState.yaw + Math.PI / 2)
            ).normalize();

            const up = new THREE.Vector3().crossVectors(right, forward).normalize();

            // Lift calculation
            const liftFactor = Math.max(0, Math.cos(flightState.pitch)) * (flightState.airspeed / 250);
            const lift = up.clone().multiplyScalar(liftFactor * 50);

            const gravity = new THREE.Vector3(0, -9.8, 0);
            const dragForce = forward.clone().multiplyScalar(flightState.airspeed * -0.1 * dragMultiplier);

            const acceleration = new THREE.Vector3()
                .add(lift)
                .add(gravity)
                .add(dragForce)
                .add(forward.multiplyScalar(flightState.airspeed * 0.5));

            flightState.velocity.add(acceleration.multiplyScalar(deltaTime * 0.001));

            const speed = flightState.velocity.length();
            if (speed > flightState.maxAirspeed * 1.5) {
                flightState.velocity.multiplyScalar((flightState.maxAirspeed * 1.5) / speed);
            }

            flightState.position.add(flightState.velocity.clone().multiplyScalar(deltaTime));

            flightState.altitude = flightState.position.y;
            flightState.verticalSpeed = flightState.velocity.y * 196.85;
            flightState.heading = ((flightState.yaw * 180 / Math.PI) % 360 + 360) % 360;

            flightState.turbulenceIntensity = Math.sin(Date.now() * 0.001) * 0.3 + Math.random() * 0.2;
            if (flightState.altitude < 10000) {
                flightState.turbulenceIntensity *= Math.max(0.5, 1 - flightState.altitude / 10000);
            }

            flightState.vibrationIntensity = (flightState.throttle / 100) * 0.005 + Math.random() * 0.002;
        }

        // WORLD OBJECTS
        let terrain;
        let cloudLayers = [];
        let skyDome;
        let sunObject;

        function createTerrain() {
            const terrainGeometry = new THREE.PlaneGeometry(200000, 200000, 128, 128);
            
            // Add terrain variation
            const positionAttribute = terrainGeometry.getAttribute('position');
            for (let i = 0; i < positionAttribute.count; i++) {
                const x = positionAttribute.getX(i);
                const z = positionAttribute.getZ(i);
                const noise = Math.sin(x * 0.0001) * Math.cos(z * 0.0001) * 500;
                positionAttribute.setY(i, noise);
            }
            positionAttribute.needsUpdate = true;

            const terrainMaterial = new THREE.MeshPhongMaterial({
                color: 0x1a4d2e,
                side: THREE.DoubleSide,
                wireframe: false,
                shininess: 30
            });
            terrain = new THREE.Mesh(terrainGeometry, terrainMaterial);
            terrain.rotation.x = -Math.PI / 2;
            terrain.position.y = -2000;
            scene.add(terrain);
        }

        function createClouds() {
            cloudLayers = [];
            const cloudAltitudes = [2000, 3500, 5500, 7500, 10000, 13000, 16000, 20000, 25000, 32000];
            
            cloudAltitudes.forEach((alt) => {
                const cloudCount = Math.floor(50 * CONFIG.cloudQuality);
                const layer = {
                    altitude: alt,
                    clouds: []
                };

                for (let i = 0; i < cloudCount; i++) {
                    const cloud = createCloudCluster(
                        Math.random() * 120000 - 60000,
                        alt,
                        Math.random() * 120000 - 60000,
                        0.4 + Math.random() * 1.8
                    );
                    layer.clouds.push(cloud);
                    scene.add(cloud);
                }

                cloudLayers.push(layer);
            });
        }

        function createCloudCluster(x, y, z, scale) {
            const group = new THREE.Group();
            const cloudCount = 5 + Math.floor(Math.random() * 6);

            for (let i = 0; i < cloudCount; i++) {
                const cloudGeometry = new THREE.SphereGeometry(
                    250 * scale + Math.random() * 300,
                    12,
                    12
                );

                const whiteVal = 0.95 + Math.random() * 0.05;
                const cloudMaterial = new THREE.MeshStandardMaterial({
                    color: new THREE.Color(whiteVal, whiteVal, whiteVal),
                    emissive: new THREE.Color(0.3, 0.3, 0.3),
                    emissiveIntensity: 0.5,
                    roughness: 0.7,
                    metalness: 0.0,
                    side: THREE.DoubleSide,
                    fog: true
                });

                const cloudMesh = new THREE.Mesh(cloudGeometry, cloudMaterial);
                cloudMesh.position.set(
                    (Math.random() - 0.5) * 1000 * scale,
                    (Math.random() - 0.4) * 500 * scale,
                    (Math.random() - 0.5) * 1000 * scale
                );
                cloudMesh.scale.set(1, 0.5 + Math.random() * 0.5, 1.3 + Math.random() * 0.3);
                cloudMesh.castShadow = true;
                cloudMesh.receiveShadow = true;
                group.add(cloudMesh);
            }

            group.position.set(x, y, z);
            group.userData = {
                baseX: x,
                baseZ: z,
                baseY: y,
                windSpeed: 2 + Math.random() * 3,
                driftOffset: Math.random() * Math.PI * 2
            };

            return group;
        }

        function createSkyDome() {
            const skyGeometry = new THREE.SphereGeometry(450000, 64, 64);
            const skyMaterial = new THREE.ShaderMaterial({
                uniforms: {
                    altitude: { value: 5000 },
                    sunDirection: { value: new THREE.Vector3(0.4, 0.7, 0.3).normalize() },
                    sunColor: { value: new THREE.Color(1, 0.95, 0.8) },
                    time: { value: 0 }
                },
                vertexShader: `
                    varying vec3 vNormal;
                    varying vec3 vPosition;
                    
                    void main() {
                        vNormal = normalize(normalMatrix * normal);
                        vPosition = (modelMatrix * vec4(position, 1.0)).xyz;
                        gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
                    }
                `,
                fragmentShader: `
                    uniform float altitude;
                    uniform vec3 sunDirection;
                    uniform vec3 sunColor;
                    uniform float time;
                    
                    varying vec3 vNormal;
                    varying vec3 vPosition;
                    
                    void main() {
                        vec3 normal = normalize(vNormal);
                        float altitudeRatio = min(altitude / 60000.0, 1.0);
                        
                        // Sky colors based on altitude
                        vec3 horizonColor = vec3(1.0, 0.85, 0.6);
                        vec3 skyLow = vec3(0.55, 0.85, 0.95);
                        vec3 skyMid = vec3(0.3, 0.6, 0.9);
                        vec3 skyHigh = vec3(0.1, 0.15, 0.5);
                        
                        vec3 finalSky = mix(horizonColor, skyLow, abs(normal.y) * 0.5);
                        finalSky = mix(finalSky, skyMid, altitudeRatio * 0.4);
                        finalSky = mix(finalSky, skyHigh, altitudeRatio * altitudeRatio * 0.8);
                        
                        // Horizon glow
                        float horizonDistance = 1.0 - abs(normal.y);
                        vec3 horizonGlow = vec3(1.0, 0.7, 0.3) * pow(horizonDistance, 8.0) * 0.3;
                        
                        // Sun position and glow
                        float sunDot = max(dot(normal, sunDirection), 0.0);
                        vec3 sunGlow = sunColor * pow(sunDot, 2.0) * 0.8;
                        
                        // Add some atmospheric banding
                        float bands = sin(normal.y * 20.0 + time * 0.1) * 0.05;
                        
                        finalSky += horizonGlow + sunGlow + bands * 0.1;
                        
                        gl_FragColor = vec4(finalSky, 1.0);
                    }
                `,
                side: THREE.BackSide,
                fog: false
            });

            skyDome = new THREE.Mesh(skyGeometry, skyMaterial);
            skyDome.position.copy(camera.position);
            scene.add(skyDome);
        }

        // LIGHTING SYSTEM
        function setupLighting() {
            const sunLight = new THREE.DirectionalLight(0xffffff, 1.3);
            sunLight.position.set(60000, 120000, 40000);
            sunLight.castShadow = CONFIG.shadowQuality > 0;
            sunLight.shadow.mapSize.width = 8192;
            sunLight.shadow.mapSize.height = 8192;
            sunLight.shadow.camera.far = 500000;
            sunLight.shadow.camera.left = -150000;
            sunLight.shadow.camera.right = 150000;
            sunLight.shadow.camera.top = 150000;
            sunLight.shadow.camera.bottom = -150000;
            sunLight.shadow.bias = -0.0001;
            scene.add(sunLight);

            const ambientLight = new THREE.AmbientLight(0x9db4ff, 0.5);
            scene.add(ambientLight);

            const skyLight = new THREE.DirectionalLight(0x7799ff, 0.4);
            skyLight.position.set(0, 100000, 0);
            scene.add(skyLight);

            const horizonLight = new THREE.DirectionalLight(0xffcc99, 0.2);
            horizonLight.position.set(100000, 20000, 100000);
            scene.add(horizonLight);
        }

        // ATMOSPHERE AND VISUALS
        function updateAtmosphere() {
            const altitudeRatio = Math.min(flightState.altitude / 60000, 1);
            
            // Dynamic fog based on altitude
            const fogNear = 8000 + altitudeRatio * 25000;
            const fogFar = 150000 + altitudeRatio * 300000;
            
            const fogHSL = {
                h: 0.6,
                s: 0.3 + altitudeRatio * 0.2,
                l: 0.55 + altitudeRatio * 0.25
            };
            
            const fogColor = new THREE.Color().setHSL(fogHSL.h, fogHSL.s, fogHSL.l);
            scene.fog = new THREE.Fog(fogColor, fogNear, fogFar);

            if (skyDome) {
                skyDome.material.uniforms.altitude.value = flightState.altitude;
                skyDome.position.copy(camera.position);
                skyDome.material.uniforms.time.value = Date.now() * 0.0001;
                
                // Dynamic sun position based on time
                const sunCycle = (Date.now() * 0.00002) % (Math.PI * 2);
                const sunDir = new THREE.Vector3(
                    Math.sin(sunCycle) * 0.6,
                    Math.cos(sunCycle) * 0.7 + 0.3,
                    0.3
                ).normalize();
                skyDome.material.uniforms.sunDirection.value = sunDir;
            }

            // Cloud opacity based on altitude
            cloudLayers.forEach(layer => {
                layer.clouds.forEach(cloud => {
                    const distToCamera = Math.abs(flightState.altitude - layer.altitude);
                    const opacity = Math.max(0.1, 1 - distToCamera / 8000);
                    
                    cloud.traverse(child => {
                        if (child.material) {
                            child.material.transparent = true;
                            child.material.opacity = opacity * 0.8;
                        }
                    });
                });
            });
        }

        // CAMERA SYSTEM
        function updateCamera() {
            const vibrationX = (Math.random() - 0.5) * flightState.vibrationIntensity * 2;
            const vibrationY = (Math.random() - 0.5) * flightState.vibrationIntensity * 2;
            const vibrationZ = (Math.random() - 0.5) * flightState.vibrationIntensity * 0.5;

            const turbulenceX = Math.sin(Date.now() * 0.002) * flightState.turbulenceIntensity * 0.3;
            const turbulenceY = Math.cos(Date.now() * 0.0015) * flightState.turbulenceIntensity * 0.15;

            const cameraPos = flightState.position.clone();
            cameraPos.x += 2 + vibrationX + turbulenceX;
            cameraPos.y += 1.5 + vibrationY + turbulenceY;
            cameraPos.z -= 1 + vibrationZ;

            camera.position.copy(cameraPos);

            const targetQuaternion = new THREE.Quaternion();
            const euler = new THREE.Euler(
                flightState.pitch + camPitch,
                flightState.yaw + camYaw,
                flightState.roll,
                'YXZ'
            );
            targetQuaternion.setFromEuler(euler);
            camera.quaternion.slerp(targetQuaternion, 0.12);
        }

        // MAIN GAME LOOP
        let gameActive = false;
        let gamePaused = false;
        let lastTime = Date.now();

        function animate() {
            requestAnimationFrame(animate);

            if (!gameActive || gamePaused) {
                renderer.render(scene, camera);
                return;
            }

            const now = Date.now();
            const deltaTime = Math.min(now - lastTime, 50);
            lastTime = now;

            updateFlightPhysics(deltaTime * 0.001);
            updateCameraLook();
            updateCamera();
            updateAtmosphere();
            updateHUD();

            // Update cloud wind animation
            cloudLayers.forEach(layer => {
                layer.clouds.forEach(cloud => {
                    const windTime = Date.now() * 0.00005;
                    cloud.position.x = cloud.userData.baseX + Math.sin(windTime + cloud.userData.driftOffset) * cloud.userData.windSpeed * 100;
                    cloud.position.z = cloud.userData.baseZ + Math.cos(windTime + cloud.userData.driftOffset) * cloud.userData.windSpeed * 80;
                });
            });

            renderer.render(scene, camera);
        }

        // INITIALIZATION
        function init() {
            camera.position.set(0, 5000, 0);
            camera.lookAt(0, 5000, 100);

            createTerrain();
            createClouds();
            createSkyDome();
            setupLighting();

            window.addEventListener('resize', () => {
                camera.aspect = window.innerWidth / window.innerHeight;
                camera.updateProjectionMatrix();
                renderer.setSize(window.innerWidth, window.innerHeight);
            });

            animate();
        }

        // START THE GAME
        init();
    </script>
