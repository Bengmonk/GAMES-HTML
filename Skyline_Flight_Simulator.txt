        // ==================== CAMERA STATE ====================
        const cameraState = {

        // ==================== THREE.JS SETUP ====================
        let scene, camera, renderer, light, ambientLight;
        let airplane, cockpit;
        let clouds = [];

        function initThreeJS() {
            const canvas = document.getElementById('canvas');
            scene = new THREE.Scene();
            
            camera = new THREE.PerspectiveCamera(
                cameraState.fov,
                window.innerWidth / window.innerHeight,
                cameraState.near,
                cameraState.far
            );
            
            renderer = new THREE.WebGLRenderer({ 
                canvas: canvas, 
                antialias: true, 
                alpha: true,
                powerPreference: 'high-performance'
            });
            renderer.setSize(window.innerWidth, window.innerHeight);
            renderer.setPixelRatio(window.devicePixelRatio);
            renderer.shadowMap.enabled = true;
            renderer.shadowMap.type = THREE.PCFShadowShadowMap;
            renderer.outputColorSpace = THREE.SRGBColorSpace;
            
            // Lighting
            light = new THREE.DirectionalLight(0xffffff, 1.2);
            light.position.set(1000, 2000, 1000);
            light.castShadow = true;
            light.shadow.mapSize.width = 4096;
            light.shadow.mapSize.height = 4096;
            light.shadow.camera.left = -5000;
            light.shadow.camera.right = 5000;
            light.shadow.camera.top = 5000;
            light.shadow.camera.bottom = -5000;
            light.shadow.camera.far = 10000;
            scene.add(light);
            
            ambientLight = new THREE.AmbientLight(0x7a9dd8, 0.6);
            scene.add(ambientLight);
            
            // Create sky
            createSky();
            
            // Create airplane
            createAirplane();
            
            // Create cockpit
            createCockpit();
            
            // Create clouds
            generateClouds();
            
            // Create ground/terrain
            createTerrain();
            
            window.addEventListener('resize', onWindowResize);

            document.addEventListener('pointerlockchange', onPointerLockChange);
        }

        // ==================== SKY SYSTEM ====================
        function createSky() {
            const skyGeometry = new THREE.SphereGeometry(250000, 32, 32);
            const skyMaterial = new THREE.MeshBasicMaterial({
                side: THREE.BackSide,
                fog: false
            });
            const sky = new THREE.Mesh(skyGeometry, skyMaterial);
            scene.add(sky);
            sky.userData.isSky = true;
            
            // Update sky color based on altitude
            updateSkyColor();
        }

        function updateSkyColor() {
            const altitude = aircraft.position.y;
            const maxAltitude = 50000;
            const altitudeRatio = Math.min(altitude / maxAltitude, 1);
            
            // Transition from light blue at sea level to dark blue at high altitude
            const lightBlue = new THREE.Color(0x87ceeb); // Sky blue
            const deepBlue = new THREE.Color(0x1a1a4d); // Deep blue
            const spaceDark = new THREE.Color(0x0d0d2b); // Space-like dark
            
            let targetColor;
            if (altitudeRatio < 0.5) {
                targetColor = lightBlue.lerp(deepBlue, altitudeRatio * 2);
            } else {
                targetColor = deepBlue.lerp(spaceDark, (altitudeRatio - 0.5) * 2);
            }
            
            scene.background = targetColor;
            
            // Update fog based on altitude
            if (altitude > 35000) {
                scene.fog = new THREE.Fog(targetColor, 100000, 200000);
            } else if (altitude > 20000) {
                scene.fog = new THREE.Fog(targetColor, 50000, 100000);
            } else if (altitude > 10000) {
                scene.fog = new THREE.Fog(targetColor, 30000, 80000);
            } else {
                scene.fog = new THREE.Fog(targetColor, 20000, 50000);
            }
            
            // Update lighting based on altitude
            if (altitude > 40000) {
                ambientLight.intensity = 0.8;
                light.intensity = 1.5;
            } else if (altitude > 30000) {
                ambientLight.intensity = 0.7;
                light.intensity = 1.3;
            } else {
                ambientLight.intensity = 0.6;
                light.intensity = 1.2;
            }
        }

        // ==================== AIRPLANE MODEL ====================
        function createAirplane() {
            airplane = new THREE.Group();
            airplane.position.copy(aircraft.position);
            scene.add(airplane);
            
            // Fuselage
            const fuselageGeometry = new THREE.CylinderGeometry(1.5, 1.2, 40, 16);
            const fuselageMaterial = new THREE.MeshStandardMaterial({
                color: 0xa0a0a0,
                metalness: 0.6,
                roughness: 0.3
            });
            const fuselage = new THREE.Mesh(fuselageGeometry, fuselageMaterial);
            fuselage.castShadow = true;
            fuselage.receiveShadow = true;
            fuselage.rotation.z = Math.PI / 2;
            airplane.add(fuselage);
            
            // Cockpit section
            const cockpitGeometry = new THREE.ConeGeometry(1.3, 8, 16);
            const cockpitMaterial = new THREE.MeshStandardMaterial({
                color: 0x2a2a3a,
                metalness: 0.7,
                roughness: 0.2
            });
            const cockpitSection = new THREE.Mesh(cockpitGeometry, cockpitMaterial);
            cockpitSection.castShadow = true;
            cockpitSection.receiveShadow = true;
            cockpitSection.position.z = 22;
            cockpitSection.rotation.z = Math.PI / 2;
            airplane.add(cockpitSection);
            
            // Wings
            const wingGeometry = new THREE.BoxGeometry(60, 2, 4);
            const wingMaterial = new THREE.MeshStandardMaterial({
                color: 0x808080,
                metalness: 0.5,
                roughness: 0.4
            });
            const wings = new THREE.Mesh(wingGeometry, wingMaterial);
            wings.castShadow = true;
            wings.receiveShadow = true;
            wings.position.y = 0;
            airplane.add(wings);
            
            // Tail
            const tailGeometry = new THREE.BoxGeometry(2, 12, 3);
            const tailMaterial = new THREE.MeshStandardMaterial({
                color: 0x7a7a8a,
                metalness: 0.5,
                roughness: 0.4
            });
            const tail = new THREE.Mesh(tailGeometry, tailMaterial);
            tail.castShadow = true;
            tail.receiveShadow = true;
            tail.position.z = -20;
            tail.position.y = 4;
            airplane.add(tail);
            
            // Engines
            for (let i = -1; i <= 1; i += 2) {
                const engineGeometry = new THREE.CylinderGeometry(1, 1, 5, 16);
                const engineMaterial = new THREE.MeshStandardMaterial({
                    color: 0x1a1a2e,
                    metalness: 0.8,
                    roughness: 0.2
                });
                const engine = new THREE.Mesh(engineGeometry, engineMaterial);
                engine.castShadow = true;
                engine.receiveShadow = true;
                engine.position.x = i * 12;
                engine.position.y = -2;
                engine.rotation.z = Math.PI / 2;
                airplane.add(engine);
            }
            
            // Landing gear (simple)
            const gearGeometry = new THREE.CylinderGeometry(0.3, 0.3, 3, 8);
            const gearMaterial = new THREE.MeshStandardMaterial({
                color: 0x1a1a1a,
                metalness: 0.9,
                roughness: 0.1
            });
            for (let i = -1; i <= 1; i += 2) {
                const gear = new THREE.Mesh(gearGeometry, gearMaterial);
                gear.castShadow = true;
                gear.position.x = i * 8;
                gear.position.y = -2;
                gear.position.z = -5;
                airplane.add(gear);
            }
        }

        // ==================== COCKPIT ====================
        function createCockpit() {
            cockpit = new THREE.Group();
            airplane.add(cockpit);
            cockpit.position.z = 18;
            
            // Cockpit frame
            const frameGeometry = new THREE.BoxGeometry(4, 2.5, 2);
            const frameMaterial = new THREE.MeshStandardMaterial({
                color: 0x1a1a2e,
                metalness: 0.3,
                roughness: 0.5
            });
            const frame = new THREE.Mesh(frameGeometry, frameMaterial);
            frame.position.z = -1;
            cockpit.add(frame);
            
            // Windshield
            const windshieldGeometry = new THREE.PlaneGeometry(4, 2, 2, 2);
            const windshieldMaterial = new THREE.MeshStandardMaterial({
                color: 0xb0e0ff,
                metalness: 0.2,
                roughness: 0.1,
                transparent: true,
                opacity: 0.3,
                side: THREE.DoubleSide
            });
            const windshield = new THREE.Mesh(windshieldGeometry, windshieldMaterial);
            windshield.position.z = 0.5;
            cockpit.add(windshield);
            
            // Side windows
            for (let side of [-1, 1]) {
                const windowGeometry = new THREE.PlaneGeometry(1.5, 1.5);

        function createDashboardTexture() {
            const canvas = document.createElement('canvas');
            canvas.width = 512;
            canvas.height = 256;
            const ctx = canvas.getContext('2d');
            
            ctx.fillStyle = '#2a2a3a';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            
            // Grid pattern
            ctx.strokeStyle = '#444455';
            ctx.lineWidth = 1;
            for (let i = 0; i < canvas.width; i += 32) {
                ctx.beginPath();
                ctx.moveTo(i, 0);
                ctx.lineTo(i, canvas.height);
                ctx.stroke();
            }
            for (let i = 0; i < canvas.height; i += 32) {
                ctx.beginPath();
                ctx.moveTo(0, i);
                ctx.lineTo(canvas.width, i);
                ctx.stroke();
            }
            
            return new THREE.CanvasTexture(canvas);
        }

        // ==================== CLOUD SYSTEM ====================
        function generateClouds() {
            clouds = [];
            
            const cloudLayers = [
                { altitude: 2000, count: 30, scale: 1.0, density: 0.6 },
                { altitude: 5000, count: 25, scale: 1.2, density: 0.5 },
                { altitude: 8000, count: 20, scale: 1.5, density: 0.4 },
                { altitude: 12000, count: 18, scale: 1.8, density: 0.35 },
                { altitude: 20000, count: 12, scale: 2.0, density: 0.3 },
                { altitude: 30000, count: 8, scale: 2.2, density: 0.25 }
            ];
            
            cloudLayers.forEach(layer => {
                for (let i = 0; i < layer.count; i++) {
                    const cloud = createCloudGroup(layer.scale, layer.density);
                    cloud.position.x = (Math.random() - 0.5) * 200000;
                    cloud.position.y = layer.altitude;
                    cloud.position.z = (Math.random() - 0.5) * 200000;
                    cloud.userData.layer = layer;
                    clouds.push(cloud);
                    scene.add(cloud);
                }
            });
        }

        function createCloudGroup(scale, density) {
            const group = new THREE.Group();
            
            const cloudMaterial = new THREE.MeshStandardMaterial({
                color: 0xffffff,
                metalness: 0,
                roughness: 0.8,
                transparent: true,
                opacity: density
            });
            
            const particleCount = Math.floor(8 * density);
            for (let i = 0; i < particleCount; i++) {
                const size = (Math.random() + 0.5) * scale * 8;
                const cloudGeometry = new THREE.SphereGeometry(size, 8, 8);
                const cloud = new THREE.Mesh(cloudGeometry, cloudMaterial);
                
                cloud.position.x = (Math.random() - 0.5) * scale * 40;
                cloud.position.y = (Math.random() - 0.5) * scale * 15;
                cloud.position.z = (Math.random() - 0.5) * scale * 40;
                
                group.add(cloud);
            }
            
            return group;
        }

        function updateClouds(deltaTime) {
            clouds.forEach(cloud => {
                cloud.position.x += Math.sin(aircraft.position.x * 0.0001) * 0.5;
                cloud.position.z += Math.cos(aircraft.position.z * 0.0001) * 0.5;
                
                // Regenerate clouds far away
                if (aircraft.position.distanceTo(cloud.position) > 150000) {
                    cloud.position.x = aircraft.position.x + (Math.random() - 0.5) * 150000;
                    cloud.position.z = aircraft.position.z + (Math.random() - 0.5) * 150000;
                }
            });
        }

        // ==================== TERRAIN ====================
        function createTerrain() {
            const terrainGeometry = new THREE.PlaneGeometry(500000, 500000, 50, 50);
            const terrainMaterial = new THREE.MeshStandardMaterial({
                color: 0x4a7c3f,
                metalness: 0,
                roughness: 0.8
            });
            
            const vertices = terrainGeometry.attributes.position.array;
            for (let i = 0; i < vertices.length; i += 3) {
                vertices[i + 2] = Math.sin(vertices[i] * 0.0001) * Math.cos(vertices[i + 1] * 0.0001) * 1000;
            }
            terrainGeometry.attributes.position.needsUpdate = true;
            terrainGeometry.computeVertexNormals();
            
            const terrain = new THREE.Mesh(terrainGeometry, terrainMaterial);
            terrain.castShadow = true;
            terrain.receiveShadow = true;
            terrain.rotation.x = -Math.PI / 2;
            terrain.position.y = -500;
            scene.add(terrain);
        }

        // ==================== FLIGHT PHYSICS ====================
        function updateFlightPhysics(deltaTime) {
            // Input
            aircraft.pitchInput = 0;
            aircraft.rollInput = 0;
            aircraft.yawInput = 0;
            
            if (keys['w']) aircraft.pitchInput -= 0.02;
            if (keys['s']) aircraft.pitchInput += 0.02;
            if (keys['a']) aircraft.rollInput -= 0.02;
            if (keys['d']) aircraft.rollInput += 0.02;
            if (keys['q']) aircraft.yawInput -= 0.02;
            if (keys['e']) aircraft.yawInput += 0.02;
            if (keys['shift']) aircraft.throttle = Math.min(aircraft.throttle + 0.01, aircraft.maxThrottle);
            if (keys['control']) aircraft.throttle = Math.max(aircraft.throttle - 0.01, aircraft.minThrottle);
            if (keys[' ']) aircraft.airbrake = true;
            else aircraft.airbrake = false;
            
            // Smoothing
            aircraft.rotation.order = 'YXZ';
            aircraft.rotation.x += aircraft.pitchInput * deltaTime * 30;
            aircraft.rotation.z += aircraft.rollInput * deltaTime * 30;
            aircraft.rotation.y += aircraft.yawInput * deltaTime * 30;
            
            // Clamp pitch
            aircraft.rotation.x = Math.max(-Math.PI / 2, Math.min(Math.PI / 2, aircraft.rotation.x));
            
            // Calculate forces
            const speed = aircraft.velocity.length();
            const speedRatio = Math.min(speed / aircraft.maxSpeed, 1);
            
            // Forward direction
            const forward = new THREE.Vector3(0, 0, -1);
            forward.applyEuler(aircraft.rotation);
            
            // Gravity
            aircraft.acceleration.y = -9.81 * 2;
            
            // Thrust
            const thrust = forward.multiplyScalar(aircraft.throttle * 100);
            aircraft.acceleration.add(thrust.multiplyScalar(1 / aircraft.mass));
            
            // Lift (simplified)
            if (speedRatio > 0.1) {
                const up = new THREE.Vector3(0, 1, 0);
                up.applyEuler(aircraft.rotation);
                const lift = up.multiplyScalar(speedRatio * 50);
                aircraft.acceleration.add(lift.multiplyScalar(1 / aircraft.mass));
            }
            
            // Drag
            const dragForce = aircraft.velocity.clone().multiplyScalar(-aircraft.dragCoefficient * speedRatio);
            aircraft.acceleration.add(dragForce.multiplyScalar(1 / aircraft.mass));
            
            // Air brake
            if (aircraft.airbrake) {
                const brakeDrag = aircraft.velocity.clone().multiplyScalar(-0.1);
                aircraft.acceleration.add(brakeDrag.multiplyScalar(1 / aircraft.mass));
            }
            
            // Update velocity and position
            aircraft.velocity.add(aircraft.acceleration.clone().multiplyScalar(deltaTime));
            aircraft.velocity.y = Math.max(aircraft.velocity.y, -100); // Terminal velocity
            aircraft.position.add(aircraft.velocity.clone().multiplyScalar(deltaTime));
            
            // Prevent underground
            if (aircraft.position.y < 0) {
                aircraft.position.y = 0;
                aircraft.velocity.y = Math.abs(aircraft.velocity.y) * 0.3;
            }
            
            // Update airplane position and rotation
            airplane.position.copy(aircraft.position);
            airplane.rotation.copy(aircraft.rotation);
            
            // Update sky
            updateSkyColor();
        }

        // ==================== CAMERA SYSTEM ====================
        function updateCamera() {
            // Apply vibration
            const engineVibration = aircraft.throttle * 0.03;
            const turbulence = (Math.sin(Date.now() * 0.002) + Math.cos(Date.now() * 0.003)) * 0.01;
            cameraState.vibration = engineVibration + turbulence;
            
            // Calculate camera position (pilot's seat in cockpit)
            const cameraPosition = new THREE.Vector3();
            cameraPosition.copy(aircraft.position);
            
            // Apply cockpit offset relative to airplane rotation
            const offset = new THREE.Vector3(
                cameraState.offsetX,
                cameraState.offsetY,
                cameraState.offsetZ
            );
            offset.applyEuler(aircraft.rotation);
            cameraPosition.add(offset);
            
            // Apply subtle vibration
            cameraPosition.x += (Math.random() - 0.5) * cameraState.vibration;
            cameraPosition.y += (Math.random() - 0.5) * cameraState.vibration;
            cameraPosition.z += (Math.random() - 0.5) * cameraState.vibration;
            
            camera.position.lerp(cameraPosition, 0.1);
            
            // Camera look direction (forward)
            const lookTarget = new THREE.Vector3(0, 0, -100);
            lookTarget.applyEuler(aircraft.rotation);
            lookTarget.add(cameraPosition);
            
            // Apply mouse look
            const yawQuaternion = new THREE.Quaternion();
            yawQuaternion.setFromAxisAngle(new THREE.Vector3(0, 1, 0), cameraState.yaw);
            
            const pitchQuaternion = new THREE.Quaternion();
            pitchQuaternion.setFromAxisAngle(new THREE.Vector3(1, 0, 0), cameraState.pitch);
            
            const lookQuaternion = new THREE.Quaternion();
            lookQuaternion.multiplyQuaternions(yawQuaternion, pitchQuaternion);
            
            const adjustedLookTarget = new THREE.Vector3(0, 0, -1);
            adjustedLookTarget.applyQuaternion(lookQuaternion);
            adjustedLookTarget.multiplyScalar(100);
            adjustedLookTarget.add(cameraPosition);
            
            camera.lookAt(adjustedLookTarget);
        }

        // ==================== INPUT HANDLING ====================
        function onKeyDown(event) {
            const key = event.key.toLowerCase();
            
            if (key === 'escape') {
                if (gameState.isFlying) {
                    togglePause();
                }
                event.preventDefault();
            }
            
            keys[key] = true;
            
            // Control key special handling
            if (event.ctrlKey) keys['control'] = true;
            if (event.shiftKey) keys['shift'] = true;
        }

        function onKeyUp(event) {
            const key = event.key.toLowerCase();
            keys[key] = false;
            if (!event.ctrlKey) keys['control'] = false;
            if (!event.shiftKey) keys['shift'] = false;
        }

        function onMouseMove(event) {
            if (!mouse.locked) return;
            
            const deltaX = event.movementX || 0;
            const deltaY = event.movementY || 0;
            
            cameraState.yaw -= deltaX * 0.003 * gameState.mouseSensitivity;
            cameraState.pitch -= deltaY * 0.003 * gameState.mouseSensitivity;
            
            // Clamp pitch
            cameraState.pitch = Math.max(-Math.PI / 2, Math.min(Math.PI / 2, cameraState.pitch));
        }

        function onPointerLockChange() {
            mouse.locked = document.pointerLockElement !== null;
        }

        // ==================== HUD UPDATE ====================
        function updateHUD() {
            const altitude = aircraft.position.y;
            const speed = aircraft.velocity.length();
            const verticalSpeed = aircraft.velocity.y;
            const heading = (aircraft.rotation.y * 180 / Math.PI + 360) % 360;
            
            document.getElementById('hudAltitude').textContent = Math.floor(altitude * 3.28084) + ' ft';
            document.getElementById('hudAirspeed').textContent = Math.floor(speed * 1.94384) + ' kts';
            document.getElementById('hudVerticalSpeed').textContent = Math.floor(verticalSpeed * 196.85) + ' fpm';
            document.getElementById('hudHeading').textContent = Math.floor(heading) + '°';
            document.getElementById('hudThrottle').textContent = Math.floor(aircraft.throttle * 100) + '%';
            document.getElementById('hudEngines').textContent = aircraft.throttle > 0.05 ? 'RUNNING' : 'IDLE';
        }

        // ==================== WINDOW RESIZE ====================
        function onWindowResize() {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }

        // ==================== MAIN ANIMATION LOOP ====================
        let lastTime = Date.now();

        function animate() {
            requestAnimationFrame(animate);
            
            const currentTime = Date.now();
            const deltaTime = (currentTime - lastTime) / 1000;
            lastTime = currentTime;
            
            if (!gameState.isPaused && gameState.isFlying) {
                updateFlightPhysics(deltaTime);
                updateClouds(deltaTime);
                updateCamera();
                updateHUD();
            }
            
            renderer.render(scene, camera);
        }
